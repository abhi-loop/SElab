# Clone repository using SSH
git clone git@github.com:username/MavenWebApp.git

# Enter repository
cd MavenWebApp

# Check status
git status

# Check remote repository
git remote -v

# Create and switch to feature branch
git switch -c feature/homepage

# Verify current branch
git branch

# Create README
touch README.md

# Stage changes
git add .

# Commit changes
git commit -m "Add homepage and README"

# Get latest main changes without losing local commits
git pull --rebase origin main

# Rebase feature branch onto main
git rebase main

# Find commit hash
git log --oneline

# Undo a commit without deleting it from history
git revert <commit-hash>

# Remove file from Git tracking but keep it locally
git rm --cached <filename>

# Commit the correction
git commit -m "Remove unwanted file from tracking"

# Compare code between branches
git diff main...feature/homepage

# Commits in feature but not main
git log main..feature/homepage

# Commits in main but not feature
git log feature/homepage..main

# Switch to main
git switch main

# Merge feature branch into main
git merge feature/homepage

# Push main to GitHub
git push origin main

# Verify synchronization
git status
